**2. POST /api/calc-plan — 计算减脂计划**
```javascript
app.post('/api/calc-plan', requireAuth, (req, res) => {
  const { current_weight, target_weight, target_date } = req.body;
  if (!current_weight || !target_weight || !target_date)
    return res.status(400).json(err(400, '请提供完整信息'));

  const profile = db.prepare('SELECT * FROM profiles WHERE user_id = ?').get(req.session.userId);
  if (!profile) return res.status(400).json(err(400, '请先完善个人资料'));

  // 计算TDEE
  const bmr = profile.gender === 'female'
    ? 10*current_weight + 6.25*profile.height - 5*profile.age - 161
    : 10*current_weight + 6.25*profile.height - 5*profile.age + 5;
  const mult = { sedentary:1.2, light:1.375, moderate:1.55, active:1.725, very_active:1.9 };
  const tdee = Math.round(bmr * (mult[profile.activity_level]||1.2));

  const daysDiff = Math.max(1, Math.ceil((new Date(target_date) - new Date())/(1000*60*60*24)));
  const weightToLose = Math.max(0, current_weight - target_weight);
  const totalDeficit = weightToLose * 7700;
  const dailyDeficit = Math.min(1000, Math.max(300, Math.round(totalDeficit/daysDiff)));
  const dailyIntake = Math.max(1200, tdee - dailyDeficit);
  const exerciseBurn = Math.round(dailyDeficit * 0.4);

  res.json(ok({
    tdee, dailyDeficit, dailyIntakeTarget: dailyIntake,
    exerciseBurnTarget: exerciseBurn, daysToTarget: daysDiff,
    weightToLose, isRealistic: dailyDeficit <= 800,
    warning: dailyDeficit > 800 ? '目标较激进，建议延长周期' : null
  }));
});
```

```javascript
// ─── 体重目标设置 ────────────────────────────────────

function openWeightGoalModal() {
  const modal = document.getElementById('weight-goal-modal');
  const cw = AppState.weightRecords[0]?.weight || AppState.profile?.weight || 70;
  document.getElementById('modal-current-weight').value = cw;
  document.getElementById('modal-target-weight').value = AppState.profile?.target_weight || '';
  document.getElementById('modal-target-date').value = AppState.profile?.target_date || '';
  modal.classList.remove('hidden');
  modal.classList.add('flex');
  if (AppState.profile?.target_weight && AppState.profile?.target_date) calcPlanPreview();
}

function closeWeightGoalModal() {
  document.getElementById('weight-goal-modal').classList.add('hidden');
  document.getElementById('weight-goal-modal').classList.remove('flex');
}

async function calcPlanPreview() {
  const cw = parseFloat(document.getElementById('modal-current-weight').value);
  const tw = parseFloat(document.getElementById('modal-target-weight').value);
  const td = document.getElementById('modal-target-date').value;
  if (!tw || !td) return;
  const res = await api('/api/calc-plan', { method:'POST', body: { current_weight:cw, target_weight:tw, target_date:td } });
  if (res?.data) {
    document.getElementById('preview-intake').textContent = res.data.dailyIntakeTarget;
    document.getElementById('preview-exercise').textContent = res.data.exerciseBurnTarget;
    document.getElementById('preview-days').textContent = `目标周期 ${res.data.daysToTarget} 天，需减${res.data.weightToLose}kg`;
    document.getElementById('plan-preview').classList.remove('hidden');
    const w = document.getElementById('preview-warning');
    res.data.warning ? (w.textContent='⚠️ '+res.data.warning, w.classList.remove('hidden')) : w.classList.add('hidden');
    AppState.planPreview = res.data;
  }
}

async function saveWeightGoal() {
  const tw = parseFloat(document.getElementById('modal-target-weight').value);
  const td = document.getElementById('modal-target-date').value;
  if (!tw || !td) return showToast('请填写完整信息', 'error');
  const data = { ...AppState.profile, target_weight:tw, target_date:td };
  const res = await api('/api/profile', { method:'PUT', body: data });
  if (res) {
    AppState.profile = { ...AppState.profile, ...data };
    updateWeightGoalDisplay();
    closeWeightGoalModal();
    showToast('目标已保存 🎯');
  }
}

function updateWeightGoalDisplay() {
  const profile = AppState.profile;
  const cw = AppState.weightRecords[0]?.weight || profile?.weight;
  if (profile?.target_weight && profile?.target_date && cw) {
    const start = AppState.weightRecords[AppState.weightRecords.length-1]?.weight || cw;
    const total = start - profile.target_weight;
    if (total <= 0) return;
    const lost = start - cw;
    const pct = Math.max(0, Math.min(100, (lost/total)*100));
    document.getElementById('target-weight-display').textContent = profile.target_weight;
    document.getElementById('target-date-display').textContent = profile.target_date;
    document.getElementById('weight-progress-bar').style.width = pct + '%';
    document.getElementById('plan-summary').textContent = `已减${lost.toFixed(1)}kg，还需减${(cw-profile.target_weight).toFixed(1)}kg 💪`;
    document.getElementById('weight-goal-progress').classList.remove('hidden');
  }
}
```

// ─── 环图渲染工具 ────────────────────────────────────
function drawRing(canvasId, value, max, color, reverseColor) {
  const canvas = document.getElementById(canvasId);
  if (!canvas) return;
  const ctx = canvas.getContext('2d');
  const dpr = window.devicePixelRatio || 1;
  const size = canvas.offsetWidth * dpr;
  canvas.width = size; canvas.height = size;
  ctx.scale(dpr, dpr);
  const ds = canvas.offsetWidth, cx = ds/2, cy = ds/2, r = ds/2 - 8, lw = 7;
  ctx.clearRect(0,0,ds,ds);
  ctx.beginPath(); ctx.arc(cx,cy,r,0,Math.PI*2);
  ctx.strokeStyle = '#f3f4f6'; ctx.lineWidth = lw; ctx.lineCap = 'round'; ctx.stroke();
  const ratio = Math.min(1, value / (max||1));
  const c = reverseColor && ratio >= 1 ? '#10b981' : color;
  ctx.beginPath(); ctx.arc(cx,cy,r,-Math.PI/2,-Math.PI/2+Math.PI*2*ratio);
  ctx.strokeStyle = c; ctx.lineWidth = lw; ctx.lineCap = 'round'; ctx.stroke();
}

function updateCalorieRings() {
  const target = AppState.planPreview?.dailyIntakeTarget || AppState.tdee - 500 || 1500;
  const exTarget = AppState.planPreview?.exerciseBurnTarget || 300;
  const intake = AppState.todayFood.reduce((s,f)=>s+f.total_calories, 0);
  const burned = AppState.todayExercise.reduce((s,e)=>s+e.calories_burned, 0);
  const remain = Math.max(0, target - intake);
  const exNeed = Math.max(0, exTarget - burned);

  drawRing('calorie-ring-main', intake, target, '#10b981');
  drawRing('calorie-ring-remaining', remain, target, '#f97316');
  drawRing('calorie-ring-exercise', burned, exTarget, '#3b82f6', true);
  drawRing('food-ring', intake, target, '#10b981');
  drawRing('exercise-ring', burned, exTarget, '#3b82f6', true);

  document.getElementById('calorie-percent').textContent = Math.round(intake/target*100)+'%';
  document.getElementById('calorie-current').textContent = Math.round(intake);
  document.getElementById('calorie-target').textContent = Math.round(target);
  document.getElementById('remaining-kcal').textContent = Math.round(remain);
  document.getElementById('exercise-need').textContent = Math.round(exNeed);
  if (document.getElementById('food-remaining-ring')) document.getElementById('food-remaining-ring').textContent = Math.round(remain);
  if (document.getElementById('food-ring-percent')) document.getElementById('food-ring-percent').textContent = Math.round(intake/target*100)+'%';
  if (document.getElementById('exercise-need-ring')) document.getElementById('exercise-need-ring').textContent = Math.round(exNeed);
  if (document.getElementById('exercise-ring-percent')) document.getElementById('exercise-ring-percent').textContent = Math.round(burned/exTarget*100)+'%';
}

## 功能三：🥗 食物适配度评分

### 评分算法（纯规则引擎，零成本）

```javascript
// ─── 食物适配度评分 ─── 加到 app.js ───

function calcFoodScore(foodName, calPer100g, amountG, mealType) {
  const p = AppState.profile;
  const todayIntake = AppState.todayFood.reduce((s,f)=>s+f.total_calories, 0);
  const target = AppState.planPreview?.dailyIntakeTarget || AppState.tdee - 500 || 1500;
  const remaining = target - todayIntake;
  const totalCal = calPer100g * (amountG||100) / 100;
  let score = 50;

  // 1. 热量密度（30分）
  score += calPer100g <= 50 ? 30 : calPer100g <= 100 ? 25 : calPer100g <= 200 ? 15 : 5;

  // 2. 时间适配（20分）
  let ts = 15;
  if (mealType === 'dinner' && calPer100g > 150) ts = 5;
  if (mealType === 'breakfast' && calPer100g > 200) ts = 8;
  if (mealType === 'snack' && calPer100g > 100) ts = 3;
  if (new Date().getHours() >= 21 && calPer100g > 100) ts = 0;
  score += ts;

  // 3. 剩余空间（30分）
  if (remaining <= 0) score -= 20;
  else if (totalCal > remaining) score -= 10;
  else if (totalCal <= remaining * 0.3) score += 10;

  // 4. 高蛋白/高油脂（20分）
  const highProtein = ['鸡胸肉','鸡蛋','牛肉','鱼肉','虾','豆腐','牛奶','蛋白粉','鸡腿','瘦猪肉'];
  const highFat = ['炸鸡','薯条','油条','蛋糕','奶茶','烧烤','火锅','肥肉','油炸'];
  if (highProtein.some(k => foodName.includes(k))) score += 15;
  if (highFat.some(k => foodName.includes(k))) score -= 15;

  score = Math.max(0, Math.min(100, score));
  let level, label, color;
  if (score >= 70)      { level='good'; label='非常适合你'; color='text-green-600 bg-green-50 border-green-200'; }
  else if (score >= 40) { level='ok';   label='还行';      color='text-yellow-600 bg-yellow-50 border-yellow-200'; }
  else                  { level='bad';  label='建议少吃';   color='text-red-600 bg-red-50 border-red-200'; }

  return { score, level, label, color, totalCal };
}

// 渲染函数
function renderFoodScore(foodName, calPer100g, amountG, mealType) {
  const r = calcFoodScore(foodName, calPer100g, amountG, mealType);
  return `<span class="inline-flex items-center gap-1 px-2 py-0.5 text-xs rounded-full border ${r.color}">${r.label} (${r.score}分)</span>`;
}
```

## 通用注意事项

1. 弹窗中 `calcPlanPreview()` 在输入变化时自动触发（oninput）
2. 环图 `drawRing` 在窗口 resize 后需要重绘
3. 食物适配度的关键词列表可根据你的食物库扩展
4. 所有新增API需 `requireAuth`
5. `planPreview` 未计算时用 `tdee-500` 和 `300` 作为默认值兜底
