## 📅 功能一：连续打卡日历 + 热力图

### 背景
当前概览页有 `check-food`、`check-exercise`、`check-water` 三个打卡图标，但只显示"今天有没有完成"，没有连续天数统计，没有日历维度的可视化，激励感极弱。

### 需求
1. **连续打卡天数**：在概览页打卡区域显示当前连续打卡天数（任意一项有记录即算打卡）
2. **月度热力图日历**：在概览页新增一个热力图日历卡片，显示当月每天的打卡状态（类似 GitHub 贡献图）
3. **打卡提醒文案**：根据连续天数显示不同激励语（1天"好的开始"、3天"初露锋芒"、7天"一周达人"、30天"习惯养成"等）

```javascript
// ─── 打卡日历 API ────────────────────────────────────

// 获取指定月份的打卡数据（每天是否有任意记录）
app.get('/api/checkin-calendar', requireAuth, (req, res) => {
  const { month } = req.query; // 格式: "2026-05"
  if (!month) return res.status(400).json(err(400, '请指定月份'));

  const startDate = `${month}-01`;
  // 计算下月第一天
  const [y, m] = month.split('-').map(Number);
  const nextMonth = m === 12 ? `${y + 1}-01` : `${y}-${String(m + 1).padStart(2, '0')}`;

  // 查询该用户当月每天是否有记录（饮食/运动/饮水任意一项）
  const rows = db.prepare(`
    SELECT DISTINCT record_date FROM (
      SELECT record_date FROM food_records WHERE user_id = ? AND record_date >= ? AND record_date < ?
      UNION
      SELECT record_date FROM exercise_records WHERE user_id = ? AND record_date >= ? AND record_date < ?
      UNION
      SELECT record_date FROM water_records WHERE user_id = ? AND record_date >= ? AND record_date < ?
    )
    ORDER BY record_date
  `).all(req.session.userId, startDate, nextMonth, req.session.userId, startDate, nextMonth, req.session.userId, startDate, nextMonth);

  // 查询连续打卡天数
  const todayStr = today();
  const checkinDates = rows.map(r => r.record_date);

  let streak = 0;
  let checkDate = todayStr;
  // 从今天往前数连续天数
  while (checkinDates.includes(checkDate)) {
    streak++;
    const d = new Date(checkDate);
    d.setDate(d.getDate() - 1);
    checkDate = d.toISOString().split('T')[0];
  }

  res.json(ok({ checkinDates, streak, today: todayStr }));
});
```

```javascript
// ─── 打卡日历 ──────────────────────────────────────
let calendarMonth = null; // 当前查看的月份

async function loadCheckinCalendar() {
  if (!calendarMonth) {
    const now = new Date();
    calendarMonth = `${now.getFullYear()}-${String(now.getMonth() + 1).padStart(2, '0')}`;
  }
  const res = await api(`/api/checkin-calendar?month=${calendarMonth}`);
  if (res && res.data) {
    AppState.checkinCalendar = res.data;
    renderCheckinCalendar();
    updateStreakDisplay(res.data.streak);
  }
}

function renderCheckinCalendar() {
  const grid = document.getElementById('calendar-grid');
  const label = document.getElementById('calendar-month-label');
  if (!grid || !AppState.checkinCalendar) return;

  const dates = AppState.checkinCalendar.dates || [];
  const [year, month] = calendarMonth.split('-').map(Number);
  label.textContent = `${year}年${month}月`;

  // 本月第一天是星期几（0=日,1=一...6=六）→ 转成周一=0的体系
  const firstDay = new Date(year, month - 1, 1).getDay();
  const offset = firstDay === 0 ? 6 : firstDay - 1; // 周一起始偏移
  const daysInMonth = new Date(year, month, 0).getDate();
  const todayStr = new Date().toISOString().split('T')[0];

  let html = '';
  // 空白占位
  for (let i = 0; i < offset; i++) {
    html += '<div></div>';
  }
  // 日期格子
  for (let d = 1; d <= daysInMonth; d++) {
    const dateStr = `${year}-${String(month).padStart(2, '0')}-${String(d).padStart(2, '0')}`;
    const checked = dates.includes(dateStr);
    const isToday = dateStr === todayStr;
    const isFuture = dateStr > todayStr;

    let bgClass = 'bg-gray-50';
    if (checked) bgClass = 'bg-primary-400';
    else if (isToday) bgClass = 'bg-primary-100 ring-2 ring-primary-400';

    html += `<div class="aspect-square flex items-center justify-center text-[11px] rounded-lg ${bgClass} ${isFuture ? 'text-gray-200' : checked ? 'text-white font-bold' : 'text-gray-500'}" title="${dateStr}">${d}</div>`;
  }
  grid.innerHTML = html;
}

function changeCalendarMonth(delta) {
  const [y, m] = calendarMonth.split('-').map(Number);
  const d = new Date(y, m - 1 + delta, 1);
  calendarMonth = `${d.getFullYear()}-${String(d.getMonth() + 1).padStart(2, '0')}`;
  loadCheckinCalendar();
}

function updateStreakDisplay(streak) {
  const countEl = document.getElementById('streak-count');
  const msgEl = document.getElementById('streak-message');
  if (countEl) countEl.textContent = streak;

  let msg = '';
  if (streak === 0) msg = '今天开始记录吧 💪';
  else if (streak === 1) msg = '好的开始！继续保持';
  else if (streak < 3) msg = '坚持就是胜利';
  else if (streak < 7) msg = '初露锋芒！连续' + streak + '天';
  else if (streak < 14) msg = '一周达人 🌟';
  else if (streak < 30) msg = '习惯正在养成！';
  else if (streak < 100) msg = '你已经是减脂达人 🔥';
  else msg = '传说级坚持者 👑';

  if (msgEl) msgEl.textContent = msg;
}
```

## 📊 功能二：周/月统计报告

### 背景
当前历史页只能按天翻看记录，用户最想看的是"这周瘦了多少""平均每天吃了多少"这种汇总视角，缺乏周/月维度的统计。

### 需求
1. **在历史Tab新增周/月切换**：支持查看"本周/本月"汇总
2. **汇总数据**：体重变化（起始→当前→变化值）、平均每日摄入、平均每日消耗、平均每日饮水、总运动次数/时长
3. **趋势图表**：周/月维度的体重折线图 + 每日热量柱状图

```javascript
// ─── 统计报告 API ────────────────────────────────────

app.get('/api/stats', requireAuth, (req, res) => {
  const { period } = req.query; // "week" 或 "month"
  const uid = req.session.userId;
  const todayStr = today();

  let startDate;
  if (period === 'month') {
    startDate = todayStr.substring(0, 7) + '-01'; // 本月1号
  } else {
    // 本周一
    const d = new Date();
    const day = d.getDay();
    const diff = day === 0 ? 6 : day - 1; // 周一=0
    d.setDate(d.getDate() - diff);
    startDate = d.toISOString().split('T')[0];
  }

  // 体重：期间首尾
  const weightRows = db.prepare(
    'SELECT weight, record_date FROM weight_records WHERE user_id = ? AND record_date >= ? AND record_date <= ? ORDER BY record_date'
  ).all(uid, startDate, todayStr);
  const startWeight = weightRows.length > 0 ? weightRows[0].weight : null;
  const endWeight = weightRows.length > 0 ? weightRows[weightRows.length - 1].weight : null;

  // 饮食：每日总摄入
  const foodRows = db.prepare(
    'SELECT record_date, SUM(total_calories) as daily_cal FROM food_records WHERE user_id = ? AND record_date >= ? AND record_date <= ? GROUP BY record_date ORDER BY record_date'
  ).all(uid, startDate, todayStr);
  const avgIntake = foodRows.length > 0
    ? Math.round(foodRows.reduce((s, r) => s + r.daily_cal, 0) / foodRows.length)
    : 0;

  // 运动：每日总消耗
  const exerciseRows = db.prepare(
    'SELECT record_date, SUM(calories_burned) as daily_cal, SUM(duration_min) as daily_min, COUNT(*) as count FROM exercise_records WHERE user_id = ? AND record_date >= ? AND record_date <= ? GROUP BY record_date ORDER BY record_date'
  ).all(uid, startDate, todayStr);
  const avgBurned = exerciseRows.length > 0
    ? Math.round(exerciseRows.reduce((s, r) => s + r.daily_cal, 0) / exerciseRows.length)
    : 0;
  const totalExerciseMin = exerciseRows.reduce((s, r) => s + (r.daily_min || 0), 0);
  const totalExerciseCount = exerciseRows.reduce((s, r) => s + (r.count || 0), 0);

  // 饮水：每日总摄入
  const waterRows = db.prepare(
    'SELECT record_date, SUM(amount_ml) as daily_ml FROM water_records WHERE user_id = ? AND record_date >= ? AND record_date <= ? GROUP BY record_date ORDER BY record_date'
  ).all(uid, startDate, todayStr);
  const avgWater = waterRows.length > 0
    ? Math.round(waterRows.reduce((s, r) => s + r.daily_ml, 0) / waterRows.length)
    : 0;

  // 打卡天数
  const checkinRows = db.prepare(`
    SELECT DISTINCT record_date FROM (
      SELECT record_date FROM food_records WHERE user_id = ? AND record_date >= ? AND record_date <= ?
      UNION
      SELECT record_date FROM exercise_records WHERE user_id = ? AND record_date >= ? AND record_date <= ?
      UNION
      SELECT record_date FROM water_records WHERE user_id = ? AND record_date >= ? AND record_date <= ?
    )
  `).all(uid, startDate, todayStr, uid, startDate, todayStr, uid, startDate, todayStr);
  const checkinDays = checkinRows.length;

  // 总天数
  const daysDiff = Math.max(Math.round((new Date(todayStr) - new Date(startDate)) / 86400000) + 1, 1);

  res.json(ok({
    period,
    startDate,
    endDate: todayStr,
    daysDiff,
    checkinDays,
    weight: { start: startWeight, end: endWeight, change: startWeight && endWeight ? +(endWeight - startWeight).toFixed(1) : null },
    avgIntake,
    avgBurned,
    avgWater,
    totalExerciseMin,
    totalExerciseCount,
    dailyFood: foodRows,       // 每日摄入数组，用于图表
    dailyExercise: exerciseRows, // 每日运动数组
    dailyWeight: weightRows,    // 每日体重数组
  }));
});
```

```javascript
let statsPeriod = 'week';

async function switchStatsPeriod(period) {
  statsPeriod = period;
  // 切换按钮样式
  ['week', 'month'].forEach(p => {
    const btn = document.getElementById(`stats-btn-${p}`);
    if (p === period) {
      btn.classList.add('bg-white', 'shadow', 'text-primary-600');
      btn.classList.remove('text-gray-500');
    } else {
      btn.classList.remove('bg-white', 'shadow', 'text-primary-600');
      btn.classList.add('text-gray-500');
    }
  });
  await loadStats();
}

async function loadStats() {
  const res = await api(`/api/stats?period=${statsPeriod}`);
  if (!res || !res.data) return;
  const d = res.data;

  // 体重变化

  if (d.weight.change !== null) {
    const change = d.weight.change;

    wcEl.className = `text-lg font-bold ${change <= 0 ? 'text-primary-600' : 'text-red-500'}`;

    wcEl.textContent = '--';
```

```javascript
let statsDailyChart = null;

function updateStatsDailyChart(foodData, exerciseData, startDate, endDate) {
  const ctx = document.getElementById('stats-daily-chart');
  if (!ctx) return;

  // 生成日期序列
  const dates = [];
  const d = new Date(startDate);
  const end = new Date(endDate);
  while (d <= end) {
    dates.push(d.toISOString().split('T')[0]);
    d.setDate(d.getDate() + 1);
  }

  const foodMap = {};
  foodData.forEach(r => foodMap[r.record_date] = Math.round(r.daily_cal));
  const exMap = {};
  exerciseData.forEach(r => exMap[r.record_date] = Math.round(r.daily_cal));

  const labels = dates.map(d => { const dt = new Date(d); return `${dt.getMonth()+1}/${dt.getDate()}`; });
  const intakeData = dates.map(d => foodMap[d] || 0);
  const burnedData = dates.map(d => exMap[d] || 0);

  if (statsDailyChart) {
    statsDailyChart.data.labels = labels;
    statsDailyChart.data.datasets[0].data = intakeData;
    statsDailyChart.data.datasets[1].data = burnedData;
    statsDailyChart.update();
    return;
  }

  statsDailyChart = new Chart(ctx, {
    type: 'bar',
    data: {
      labels,
```

```javascript
// ─── 定时提醒 ──────────────────────────────────────

let reminderTimers = [];

async function requestNotificationPermission() {
  if (!('Notification' in window)) {
    showToast('浏览器不支持通知功能', 'warning');
    return;
  }
  const permission = await Notification.requestPermission();
  if (permission === 'granted') {
    showToast('通知权限已开启');
    document.getElementById('notification-permission')?.classList.add('hidden');
  } else {
    showToast('通知权限被拒绝', 'error');
  }
}

function checkNotificationPermission() {
  if (!('Notification' in window) || Notification.permission === 'denied') {
    document.getElementById('notification-permission')?.classList.remove('hidden');
  }
}

function fillReminderForm() {
  if (!AppState.profile) return;
  const p = AppState.profile;
  const weighTime = document.getElementById('reminder-weigh-time');
  const weighEnabled = document.getElementById('reminder-weigh-enabled');
  const waterInterval = document.getElementById('reminder-water-interval');
  const waterEnabled = document.getElementById('reminder-water-enabled');
  const exerciseTime = document.getElementById('reminder-exercise-time');
  const exerciseEnabled = document.getElementById('reminder-exercise-enabled');

  if (weighTime) weighTime.value = p.reminder_weigh || '08:00';
  if (weighEnabled) weighEnabled.checked = !!p.reminder_weigh_enabled;
  if (waterInterval) waterInterval.value = p.reminder_water_interval || '120';
  if (waterEnabled) waterEnabled.checked = !!p.reminder_water_enabled;
  if (exerciseTime) exerciseTime.value = p.reminder_exercise || '18:00';
  if (exerciseEnabled) exerciseEnabled.checked = !!p.reminder_exercise_enabled;

  // 启动定时器
  startReminders();
}

async function saveReminders() {
  if (!AppState.profile) return;
  const data = {
    ...AppState.profile,
    reminder_weigh: document.getElementById('reminder-weigh-time')?.value || '08:00',
    reminder_weigh_enabled: document.getElementById('reminder-weigh-enabled')?.checked ? 1 : 0,
    reminder_water_interval: parseInt(document.getElementById('reminder-water-interval')?.value) || 120,
    reminder_water_enabled: document.getElementById('reminder-water-enabled')?.checked ? 1 : 0,
    reminder_exercise: document.getElementById('reminder-exercise-time')?.value || '18:00',
    reminder_exercise_enabled: document.getElementById('reminder-exercise-enabled')?.checked ? 1 : 0,
  };
  const res = await api('/api/profile', { method: 'PUT', body: data });
  if (res) {
    AppState.profile = { ...AppState.profile, ...data };
    startReminders(); // 重新设置定时器
    showToast('提醒设置已保存');
  }
}

function startReminders() {
  // 清除旧定时器
  reminderTimers.forEach(t => clearInterval(t));
  reminderTimers = [];

  const p = AppState.profile;
  if (!p) return;

  // 每分钟检查一次是否到了提醒时间
  const checkInterval = setInterval(() => {
    const now = new Date();
    const hh = String(now.getHours()).padStart(2, '0');
    const mm = String(now.getMinutes()).padStart(2, '0');
    const timeStr = `${hh}:${mm}`;

    // 称重提醒
    if (p.reminder_weigh_enabled && timeStr === p.reminder_weigh) {
      sendNotification('⚖️ 称重提醒', '该记录今天的体重啦！');
    }

    // 运动提醒
    if (p.reminder_exercise_enabled && timeStr === p.reminder_exercise) {
      sendNotification('🏃 运动提醒', '该运动啦，动起来！');
    }
  }, 60000); // 每分钟检查

  reminderTimers.push(checkInterval);

  // 喝水提醒（按间隔）
  if (p.reminder_water_enabled) {
    const intervalMs = (p.reminder_water_interval || 120) * 60 * 1000;
    const waterTimer = setInterval(() => {
      sendNotification('💧 喝水提醒', '记得喝水，保持水分充足！');
    }, intervalMs);
    reminderTimers.push(waterTimer);
  }
}

function sendNotification(title, body) {
  if (!('Notification' in window)) return;
  if (Notification.permission === 'granted') {
    new Notification(title, { body, icon: '/favicon.ico' });
  }
}
```
